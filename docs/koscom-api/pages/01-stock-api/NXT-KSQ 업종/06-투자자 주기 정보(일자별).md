      payload = {
        "cust_id" : "NS00000001",

        "data_list" : "F16013, F15001",
        "jcode" : "51",
        "edate" : "20231231"
      }
      r = session.post(host_url + api_url, data = payload)

      # 정렬. ensure_ascii = 한글 깨짐 방지. indent = 들여쓰기.
      print(json.dumps(r.json(), ensure_ascii=False, indent=2))
